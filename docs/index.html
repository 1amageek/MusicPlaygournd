var body: some Sound {
    RhythmSection()

    // Shape the filter live.
    SynthSection(
        acidAmount: slider(0.5, in: 0...1)
    )

    BusReturn("synths")
        .gain(slider($synthLevel, in: 0...1))
}

Get started 
git clone --branch 0.5.1 https://github.com/1amageek/MusicPlaygournd.git
cd MusicPlaygournd
./Scripts/build-app.sh
open .build/MusicPlaygournd.app

This preview is built locally from source. Keep your Swift compiler and SDK installed. Initial package resolution and compilation can take longer; progress appears in the sidebar.
